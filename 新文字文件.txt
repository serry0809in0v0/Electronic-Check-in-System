# 1. 使用者簽到
if menu == "使用者簽到":
  st.subheader("📝 每日快速簽到")
  with st.form("checkin_form"):
    col1, col2 = st.columns(2)
    with col1:
      user_id = st.selectbox(
          "選擇使用者帳號",
          list(st.session_state["users"].keys()),
          format_func=lambda x: (
              f"{x} - {st.session_state['users'][x]['name']}"
          ),
      )
      pin = st.text_input("輸入 4 位數 PIN 碼驗證", type="password")
    with col2:
      mood = st.text_input(
          "今日心情或目標寫真", placeholder="例：今天目標完成專案報告！"
      )
      weather_input = st.selectbox("確認今日天氣", ["晴天", "多雲", "雨天"])

    submit = st.form_submit_button("立即簽到")
    if submit:
      user_info = st.session_state["users"][user_id]
      if pin == user_info["pin"]:
        today = datetime.date.today().strftime("%Y-%m-%d")
        df_logs = st.session_state["logs"]
        already_checked = not df_logs[
            (df_logs["日期"] == today) & (df_logs["帳號"] == user_id)
        ].empty

        if already_checked:
          st.warning("⚠️ 您今天已經完成簽到過了，請勿重複簽到！")
        else:
          new_row = {
              "日期": today,
              "帳號": user_id,
              "姓名": user_info["name"],
              "時間": datetime.datetime.now().strftime("%H:%M"),
              "天氣": weather_input,
              "心情": mood,
              "類型": "正常",
          }
          st.session_state["logs"] = pd.concat(
              [df_logs, pd.DataFrame([new_row])], ignore_index=True
          )

          # 更新連續天數與獎勵
          user_info["streak"] += 1
          if (
              user_info["streak"] == 7
              and "七天連續簽到" not in user_info["badges"]
          ):
            user_info["badges"].append("七天連續簽到 🏅")
            st.balloons()

          st.success(
              f"🎉 簽到成功！目前連續簽到天數：{user_info['streak']}"
              f" 天。獲得徽章：{', '.join(user_info['badges'])}"
          )
      else:
        st.error("❌ PIN 碼錯誤，驗證失敗！")

# 2. 補簽功能
elif menu == "補簽功能":
  st.subheader("🔄 缺席紀錄補簽")
  user_id = st.selectbox(
      "選擇使用者帳號",
      list(st.session_state["users"].keys()),
      format_func=lambda x: f"{x} - {st.session_state['users'][x]['name']}",
  )
  user_info = st.session_state["users"][user_id]

  st.info(
      f"使用者：{user_info['name']} | 剩餘補簽次數："
      f" **{user_info['makeup_left']}** 次"
  )

  with st.form("makeup_form"):
    makeup_date = st.date_input(
        "選擇要補簽的日期",
        max_value=datetime.date.today() - datetime.timedelta(days=1),
    )
    makeup_reason = st.text_input("補簽原因與心情備註")
    submit_makeup = st.form_submit_button("確認補簽")

    if submit_makeup:
      if user_info["makeup_left"] > 0:
        date_str = makeup_date.strftime("%Y-%m-%d")
        df_logs = st.session_state["logs"]
        exists = not df_logs[
            (df_logs["日期"] == date_str) & (df_logs["帳號"] == user_id)
        ].empty
        if exists:
          st.warning("該日已有簽到紀錄，無需補簽。")
        else:
          user_info["makeup_left"] -= 1
          new_row = {
              "日期": date_str,
              "帳號": user_id,
              "姓名": user_info["name"],
              "時間": "12:00 (補簽)",
              "天氣": "晴天",
              "心情": makeup_reason,
              "類型": "補簽",
          }
          st.session_state["logs"] = pd.concat(
              [df_logs, pd.DataFrame([new_row])], ignore_index=True
          )
          st.success(
              f"✅ 成功補簽 {date_str}！剩餘補簽次數："
              f" {user_info['makeup_left']} 次"
          )
      else:
        st.error("❌ 您的補簽次數已用完，無法補簽！")

# 3. 統計圖表與日曆視圖
elif menu == "統計圖表與日曆視圖":
  st.subheader("📊 簽到統計與視覺化")
  df_logs = st.session_state["logs"]

  if df_logs.empty:
    st.info("目前尚無簽到紀錄。")
  else:
    col1, col2 = st.columns(2)
    with col1:
      st.markdown("#### 每日簽到人數統計")
      daily_counts = df_logs.groupby("日期").size()
      fig, ax = plt.subplots(figsize=(6, 4))
      daily_counts.plot(kind="bar", color="#4CAF50", ax=ax)
      ax.set_ylabel("簽到人數")
      ax.set_xlabel("日期")
      plt.xticks(rotation=45)
      st.pyplot(fig)

    with col2:
      st.markdown("#### 使用者連續簽到排行")
      streak_data = {
          uid: info["streak"] for uid, info in st.session_state["users"].items()
      }
      streak_df = pd.DataFrame(
          list(streak_data.items()), columns=["帳號", "連續天數"]
      ).sort_values(by="連續天數", ascending=False)
      st.dataframe(streak_df, use_container_width=True)

    st.markdown("### 📋 所有簽到紀錄與心情清單")
    st.dataframe(df_logs, use_container_width=True)
